import { PropertyInquiry, useSubmitPropertyInquiry } from "@workspace/api-client-react";
import { z } from "zod";
import { FormEvent, useState } from "react";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Textarea } from "@/components/ui/textarea";

const defaultMessage = "I'm interested in this property.";
const inquiryFormSchema = z.object({
  name: z.string().trim().min(2, "Name must be at least 2 characters.").max(100),
  email: z.string().trim().email("Enter a valid email address.").max(254),
  phone: z.string().trim().max(30, "Phone number is too long.").optional(),
  message: z.string().trim().min(1, "Message is required.").max(2000),
});

export function PropertyInquiryForm({
  listingKey,
  listingId,
}: Pick<PropertyInquiry, "listingKey" | "listingId">) {
  const [form, setForm] = useState({ name: "", email: "", phone: "", message: defaultMessage });
  const [validationError, setValidationError] = useState<string>();
  const mutation = useSubmitPropertyInquiry();

  const update = (field: keyof typeof form, value: string) => {
    setForm((current) => ({ ...current, [field]: value }));
    setValidationError(undefined);
  };

  const submit = async (event: FormEvent<HTMLFormElement>) => {
    event.preventDefault();
    setValidationError(undefined);
    const parsed = inquiryFormSchema.safeParse({
      name: form.name,
      email: form.email,
      phone: form.phone || undefined,
      message: form.message,
    });
    if (!parsed.success) {
      setValidationError(parsed.error.issues[0]?.message ?? "Please check your inquiry details.");
      return;
    }
    await mutation.mutateAsync({ data: { listingKey, listingId, ...parsed.data } });
  };

  if (mutation.isSuccess) {
    return (
      <div className="rounded-lg border border-emerald-200 bg-emerald-50 p-5 text-sm text-emerald-900 dark:border-emerald-900 dark:bg-emerald-950/30 dark:text-emerald-200">
        <p className="font-semibold">Your inquiry was received.</p>
        <p className="mt-1">Michael will receive your inquiry and can follow up using the contact details you provided.</p>
      </div>
    );
  }

  return (
    <form onSubmit={submit} className="space-y-4 rounded-lg border border-border bg-card p-5">
      <div>
        <h3 className="text-xl font-serif font-medium">Ask about this property</h3>
        <p className="mt-1 text-sm text-muted-foreground">Send an inquiry to Michael about this property. No account is required.</p>
      </div>
      <div className="grid gap-4 sm:grid-cols-2">
        <label className="space-y-1.5 text-sm font-medium">
          Name <span className="text-destructive">*</span>
          <Input value={form.name} onChange={(event) => update("name", event.target.value)} autoComplete="name" required />
        </label>
        <label className="space-y-1.5 text-sm font-medium">
          Email <span className="text-destructive">*</span>
          <Input type="email" value={form.email} onChange={(event) => update("email", event.target.value)} autoComplete="email" required />
        </label>
      </div>
      <label className="block space-y-1.5 text-sm font-medium">
        Phone <span className="font-normal text-muted-foreground">(optional)</span>
        <Input type="tel" value={form.phone} onChange={(event) => update("phone", event.target.value)} autoComplete="tel" />
      </label>
      <label className="block space-y-1.5 text-sm font-medium">
        Message <span className="text-destructive">*</span>
        <Textarea value={form.message} onChange={(event) => update("message", event.target.value)} rows={4} required />
      </label>
      {(validationError || mutation.isError) && (
        <p role="alert" className="text-sm text-destructive">
          {validationError ?? "We could not receive your inquiry. Please try again."}
        </p>
      )}
      <Button type="submit" disabled={mutation.isPending}>
        {mutation.isPending ? "Sending inquiry..." : "Send inquiry"}
      </Button>
      <p className="text-xs text-muted-foreground">Your inquiry is sent securely to the site owner. Your email address is used as the reply-to address.</p>
    </form>
  );
}
