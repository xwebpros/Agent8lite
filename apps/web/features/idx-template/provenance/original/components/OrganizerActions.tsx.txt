import { addMyComparisonItem, createMySavedSearch, getMyComparison, type ListListingsParams, type SavedSearchCriteria } from "@workspace/api-client-react";
import { useState } from "react";
import { Link } from "wouter";

function criteriaFromFilters(filters: ListListingsParams): SavedSearchCriteria {
  const { sort: _sort, page: _page, ...criteria } = filters;
  return criteria;
}

export function SaveSearchButton({ filters }: { filters: ListListingsParams }) {
  const [message, setMessage] = useState("");
  async function save() {
    const name = window.prompt("Name this search")?.trim();
    if (!name) return;
    try { await createMySavedSearch({ name, criteria: criteriaFromFilters(filters) }); setMessage("Saved"); } catch { setMessage("Sign in or use a new search name"); }
  }
  return <span className="inline-flex items-center gap-2"><button type="button" onClick={() => void save()} className="rounded border px-3 py-2 text-sm hover:border-primary">Save this search</button>{message === "Saved" ? <Link href="/account/organizer" className="text-xs text-primary underline">View saved</Link> : message ? <span role="status" className="text-xs text-muted-foreground">{message}</span> : null}</span>;
}

export function CompareButton({ listingKey }: { listingKey: string }) {
  const [message, setMessage] = useState("");
  async function add() {
    try { const current = await getMyComparison(); await addMyComparisonItem(current.comparison.id, listingKey); setMessage("Added"); } catch { setMessage("Sign in to compare"); }
  }
  return <button type="button" onClick={() => void add()} className="rounded border bg-background px-2 py-1 text-xs hover:border-primary" aria-label={`Compare property ${listingKey}`}>{message || "Compare"}</button>;
}
