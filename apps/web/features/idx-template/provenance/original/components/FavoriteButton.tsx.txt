import { Heart } from "lucide-react";
import { useEffect, useState } from "react";
import { getFavoriteKeys, persistFavorite, subscribeToAuthChanges, subscribeToFavorites, syncFavoritesWithServer } from "@/lib/favorites";

export function FavoriteButton({ listingKey }: { listingKey: string }) {
  const [isFavorite, setIsFavorite] = useState(() => getFavoriteKeys().includes(listingKey));
  const [pending, setPending] = useState(false);
  const [error, setError] = useState("");

  useEffect(() => {
    let cancelled = false;
    const refresh = () => setIsFavorite(getFavoriteKeys().includes(listingKey));
    void syncFavoritesWithServer().then(() => { if (!cancelled) refresh(); }).catch(() => undefined);
    const unsubscribeFavorites = subscribeToFavorites(refresh);
    const unsubscribeAuth = subscribeToAuthChanges(() => {
      void syncFavoritesWithServer().then(() => { if (!cancelled) refresh(); }).catch(() => undefined);
    });
    return () => { cancelled = true; unsubscribeFavorites(); unsubscribeAuth(); };
  }, [listingKey]);

  const toggleFavorite = async () => {
    const next = !isFavorite;
    setIsFavorite(next);
    setPending(true);
    setError("");
    try {
      await persistFavorite(listingKey, next);
    } catch (cause) {
      setIsFavorite(!next);
      setError(cause instanceof Error ? cause.message : "Unable to update this saved home.");
    } finally {
      setPending(false);
    }
  };

  return (
    <button
      type="button"
      aria-label={isFavorite ? "Remove from favorites" : "Add to favorites"}
      aria-pressed={isFavorite}
      disabled={pending}
      title={error || (isFavorite ? "Remove from favorites" : "Add to favorites")}
      onClick={(event) => {
        event.preventDefault();
        event.stopPropagation();
        void toggleFavorite();
      }}
      className="inline-flex h-10 w-10 items-center justify-center rounded-full border border-border bg-background/95 text-foreground shadow-sm transition hover:bg-background focus:outline-none focus:ring-2 focus:ring-primary"
    >
      <Heart className={`h-5 w-5 ${isFavorite ? "fill-primary text-primary" : ""}`} />
      {error && <span className="sr-only" role="alert">{error}</span>}
    </button>
  );
}
